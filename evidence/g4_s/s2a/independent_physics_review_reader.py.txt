No Torch, actor, MuJoCo, model compilation, physics, or experiment entry point is
imported. Raw JSONL/NPZ and frozen S1B evidence are the only numeric inputs.
"""

def load(path):
    return json.loads(path.read_text())

def rows(path):
    return [json.loads(line) for line in path.read_text().splitlines()]

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def matrix(quaternion):
    w, x, y, z = quaternion
    return np.array([[1-2*(y*y+z*z), 2*(x*y-z*w), 2*(x*z+y*w)],
                     [2*(x*y+z*w), 1-2*(x*x+z*z), 2*(y*z-x*w)],
                     [2*(x*z-y*w), 2*(y*z+x*w), 1-2*(x*x+y*y)]])

def relative(state):
    hand = matrix(state['tcp_quaternion_wxyz'])
    peg = matrix(state['peg_quaternion_wxyz'])
    position = hand.T @ (np.asarray(state['peg_root_position']) - state['tcp_position_world'])
    return position, hand.T @ peg

def angle(a, b):
    return float(math.acos(np.clip((np.trace(a.T @ b)-1)/2, -1, 1)))

def pad_forces(state):
    forces = {'left_finger': 0., 'right_finger': 0.}
    for contact in state['contacts']:
        if 'peg_collision' not in (contact['geom1'], contact['geom2']):
            continue
        other = contact['geom2'] if contact['geom1'] == 'peg_collision' else contact['geom1']
        for side in forces:
            if other in {side + '_geom_' + str(k) for k in range(3, 8)}:
                forces[side] += max(0., contact['normal_force_n'])
    return forces

def limits(q, bounds):
    return np.maximum(np.maximum(bounds[:, 0]-q, q-bounds[:, 1]), 0.)

def witness(row, values=None):
    return {'policy_step': row['policy_step'], 'episode_physics_step': row['physics_step'],
            'episode_elapsed_simulation_s': row['physics_step']/120,
            'absolute_MuJoCo_data_time_s': row['state']['time_s'],
            'values': values, 'state': row['state']}
